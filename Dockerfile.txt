# Usar una imagen base ligera de Node.js
FROM node:18-bullseye-slim

# Evitar prompts interactivos durante la instalación
ENV DEBIAN_FRONTEND=noninteractive

# Instalar Python, dependencias de compilación y Chromium (Chrome para Linux)
RUN apt-get update && apt-get install -y \
    python3 \
    python3-pip \
    chromium \
    libgconf-2-4 \
    libatk1.0-0 \
    libatk-bridge2.0-0 \
    libgdk-pixbuf2.0-0 \
    libgtk-3-0 \
    libgbm-dev \
    libnss3-dev \
    libxss-dev \
    && rm -rf /var/lib/apt/lists/*

# Instalar ddddocr (la red neuronal del Captcha)
RUN pip3 install ddddocr

# Evitar que Puppeteer descargue su propio Chrome pesado (usaremos el Chromium ligero del sistema)
ENV PUPPETEER_SKIP_CHROMIUM_DOWNLOAD=true
ENV PUPPETEER_EXECUTABLE_PATH=/usr/bin/chromium

# Configurar directorio de trabajo
WORKDIR /app

# Copiar archivos de dependencias y código
COPY package*.json ./
RUN npm install --production
COPY . .

# Exponer el puerto (Render usa el puerto 10000 por defecto o el que definas)
EXPOSE 3000

# Comando para iniciar la app
CMD ["npm", "start"]
